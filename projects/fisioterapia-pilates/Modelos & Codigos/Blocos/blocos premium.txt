```js
gsap.registerPlugin(ScrollTrigger);
const root = document.querySelector('[data-effect="atelier-card-scroll"]');
const scrollShell = root.closest('.scroll-shell');
const cardContainers = gsap.utils.toArray(root.querySelectorAll('.card-container'));
const cardFaces = cardContainers.map((card) => card.querySelector('.card-3d'));
const progressBars = root.querySelectorAll('.ui-progress i');
let masterTimeline;

function splitText(node) {
  if (node.dataset.splitDone) return;
  const text = node.textContent;
  node.textContent = '';
  for (const char of text) {
    const span = document.createElement('span');
    span.className = 'text-char';
    span.textContent = char === ' ' ? '\u00a0' : char;
    node.appendChild(span);
  }
  node.dataset.splitDone = 'true';
}

root.querySelectorAll('.headline-foreground [data-split-text]').forEach(splitText);
const foregroundChars = root.querySelectorAll('.headline-foreground .text-char');

function updateProgress() {
  const max = window.innerWidth * 0.6 || 1;
  const values = cardContainers.map((card) => {
    const distance = Math.abs(Number(gsap.getProperty(card, 'x')));
    return Math.max(0, Math.min(12, 12 - (distance / max) * 12));
  });
  gsap.set(progressBars[0], { width: 3 + values[0] - values[1], opacity: .5 + values[0] / 24 - values[1] / 24 });
  gsap.set(progressBars[1], { width: 3 + values[1] - values[2], opacity: .5 + values[1] / 24 - values[2] / 24 });
  gsap.set(progressBars[2], { width: 3 + values[2], opacity: .5 + values[2] / 24 });
}

function setupAnimation() {
  masterTimeline?.scrollTrigger?.kill();
  masterTimeline?.kill();
  const xStart = [window.innerWidth * .6, window.innerWidth * -.6, window.innerWidth * .6];
  const zRotation = [-3.5, 9.5, -9.6];
  const rotationY = [90, -90, 90];
  const zStart = Math.max(500, 500 / (1440 / window.innerWidth));
  const perspective = Math.max(1000, 1000 / (1440 / window.innerWidth));
  cardContainers.forEach((card) => { card.style.perspective = `${perspective}px`; });
  gsap.set(cardContainers, { xPercent: -50, yPercent: -50, x: i => xStart[i], rotation: i => zRotation[i], autoAlpha: 0, transformOrigin: '50% 50%' });
  gsap.set(cardFaces, { rotationY: i => rotationY[i], z: zStart, transformOrigin: '50% 50%' });
  gsap.set(progressBars, { width: 3, opacity: .5 });
  masterTimeline = gsap.timeline({ scrollTrigger: { trigger: scrollShell, start: 'top top', end: 'bottom bottom', scrub: 1, pin: false, invalidateOnRefresh: true, onUpdate: updateProgress } });
  masterTimeline
    .to(cardContainers[0], { x: 0, autoAlpha: 1, duration: 1.35, ease: 'power2.inOut' }, 0)
    .to(cardFaces[0], { rotationY: 0, z: 0, duration: 1.35, ease: 'power2.inOut' }, 0)
    .to(cardContainers[1], { x: 0, autoAlpha: 1, duration: 1.35, ease: 'power2.inOut' }, .85)
    .to(cardFaces[1], { rotationY: 0, z: 0, duration: 1.35, ease: 'power2.inOut' }, .85)
    .to(cardContainers[2], { x: 0, autoAlpha: 1, duration: 1.35, ease: 'power2.inOut' }, 1.70)
    .to(cardFaces[2], { rotationY: 0, z: 0, duration: 1.35, ease: 'power2.inOut' }, 1.70)
    .to(foregroundChars, { opacity: 0, duration: 1.6, stagger: .05, ease: 'power2.inOut' }, .30);
  ScrollTrigger.refresh();
}
```

- Assign each slide to a layer by `index % 3`.
- Base slide rotation:
  - Even indexes rotate negative.
  - Odd indexes rotate positive.
  - Magnitude is random per setup: `Math.random() * 3 + 2`, giving roughly 2 to 5 degrees.
- On every animation frame while active:
  - Increase `current` by `0.5`.
  - Lerp layer 0 toward `current` with `0.09 * ratio`.
  - Lerp layer 1 toward `current` with `0.1 * ratio`.
  - Lerp layer 2 toward `current` with `0.11 * ratio`.
  - Compute `velocityOffset = layers[0] - current`.
  - Round `velocityOffset` to three decimals.
  - Render cards unless resizing.
- Frame ratio:
  - `ratio = Math.max(0.25, Math.min(4, (time - lastTime) / 16.6667))`
- Slide transform:
  - Use `translate3d(-xpx, 0, 0) rotate(baseRotation + velocityTilt deg)`.
  - `velocityTilt` is based on `velocityOffset * 0.03`.
  - Cap tilt smoothly with:
    `Math.sign(rawTilt) * 10 * (1 - Math.exp(-Math.abs(rawTilt) / 10))`
- Wrapping:
  - Use `gsap.utils.wrap(-(maxOffset - item.end), item.end, layers[item.layer])`.
  - Only render visible cards or cards transitioning out, using a visibility test `x > start && x < end`.
- ScrollTrigger:
  - Create one `ScrollTrigger` per section.
  - Set `active` in `onToggle` based on `isActive`.
  - While inactive, do not auto-increment `current` and do not respond to wheel movement.
- Wheel behavior:
  - Listen to `wheel` on the section with `{ passive: true }`.
  - If active, add `event.deltaY * 0.5` to `current`.
- Drag behavior:
  - `mousedown` and `touchstart` call pointer down.
  - `mousemove` and `touchmove` call pointer move.
  - `mouseup` and `touchend` on `window` call pointer up.
  - On pointer down, add `.is-dragging`, store pointer start coordinates, and set `dragStartScroll = current + pointerX * speedMultiplier`.
  - On pointer move, if horizontal movement is greater than vertical movement and the event is cancelable, call `preventDefault()` and `stopPropagation()`.
  - Set `current = dragStartScroll - pointerX * speedMultiplier`.
  - On pointer up, remove `.is-dragging`.
  - If horizontal movement is at least `10px`, snap to the nearest slide.
- Drag speed:
  - Desktop/tablet at `min-width: 650px`: `speedMultiplier = 2`
  - Small screens below `650px`: `speedMultiplier = 3.5`
- Snap behavior:
  - Wrap `current` into `[0, maxOffset]`.
  - Use `gsap.utils.snap(snapPoints, wrapped)`.
  - Add `target - wrapped` to `current`.
- Resize behavior:
  - On resize, rerun setup.
  - During setup, set `resizing = true`; after one animation frame, set `resizing = false`.
  - Recalculate slide rects, snap points, `maxOffset`, base rotations, layers, and ScrollTrigger.
- Cleanup behavior:
  - Remove all event listeners.
  - Kill ScrollTrigger.
  - Cancel RAF.
  - Remove cloned slides.

INTERACTIONS AND MOTION
- Register GSAP plugin with `gsap.registerPlugin(ScrollTrigger)`.
- Initialize the effect with a function named `initMethodScrollAccordion(root = document)`.
- Query sections with `[data-effect="method-scroll-accordion"]`.
- For each section:
  - Query all `.method-item` cards.
  - Query `.method-title`, `.method-progress`, `.method-progress-fill`, and `.method-glow`.
  - Build a paused GSAP timeline and control it from ScrollTrigger progress.
- ScrollTrigger setup:
  - `id: "method-scroll-accordion"`
  - `trigger: section`
  - `start: "top top"`
  - `end: () => "+=" + (window.innerHeight * items.length)`
  - `pin: section.querySelector(".method-wrapper")`
  - `pinSpacing: true`
  - `anticipatePin: 1`
  - `scrub: true`
  - `invalidateOnRefresh: true`
- Timeline setup:
  - Before animating, measure each `.method-body` by setting height to `auto`, reading `scrollHeight`, then resetting height to `0`.
  - Each item opens from height `0` to its measured height over `0.4s` with `ease: "linear"` and `immediateRender: false`.
  - Each item then holds active visual state for `0.6s`.
  - Every item except the last closes from measured height back to `0` over `0.4s` with `ease: "linear"` and `immediateRender: false`.
  - The last item receives a final empty `0.3s` hold instead of closing immediately.
- Scroll update behavior:
  - Desktop title y-position maps section progress `0..1` to `50%..-50%`.
  - Mobile title must not receive JS transform updates; guard with `!window.matchMedia("(max-width: 768px)").matches`.
  - Timeline progress maps ScrollTrigger progress `0..1` to `0.05..1`.
  - Outer progress rail scale maps early scroll progress `0..0.15` to `scaleX(0.5)..scaleX(1)`.
  - Progress fill uses `scaleX(progress)`.
  - Glow y offset maps progress `0..1` to `-12vh..16vh`.
  - Glow scale maps progress `0..1` to `0.8..1.2`.
  - Glow opacity maps progress `0..1` to `0.55..1`.
- Item visual update behavior:
  - Derive `openRatio`: if `ratio <= 1`, use `ratio`; if `ratio <= 1.5`, use `1`; otherwise use `2.5 - ratio`.
  - Clamp active amount to `0..1`.
  - Gradient progress maps capped ratio `0..1.3` to `0..1`.
  - While opening, gradient opacity maps `0..1` to `0..1`; while closing, opacity maps ratio `1.5..2.5` to `1..0`.
  - Gradient y-percent maps `0..1` to `20..-60`.
  - Gradient scale interpolates `1..1.5`.
  - White overlay opacity maps opening `0.8..1` to `0..0.7`, and closing `1.5..2.5` to `0.7..0`.
  - Quote opacity maps active amount `0.3..1` to `0..1`; quote y maps `28px..0`.
  - Signature opacity equals active amount; signature y-percent maps active amount `0.23..0.3` to `100..0`.
  - Toggle `.method-logo.is-active` when active amount is greater than `0.5`.
- Resize behavior:
  - Use `gsap.delayedCall(0.15, setup).pause()` to debounce resize.
  - On resize, rebuild measurements, timeline, and ScrollTrigger.
  - Return a cleanup function that removes the resize listener, kills the delayed call, kills ScrollTrigger, and kills the timeline.
- Preview-only behavior:
  - On `DOMContentLoaded`, call `initMethodScrollAccordion()`.
  - If query parameter `preview-progress` exists, refresh ScrollTrigger and scroll to the matching trigger progress. This is only for screenshot/preview control and can be omitted in a production component.

INTERACTIONS AND MOTION
- Register GSAP plugins with `gsap.registerPlugin(ScrollTrigger, Draggable)`.
- Initialize text reveal with `initLineReveal(root = document)`.
- Initialize cards with `initFlickCardsStack(root = document)`.

Text reveal behavior:
- Query all `[data-split="heading"]`.
- Skip any element with `dataset.revealReady === "true"`.
- Save the original trimmed text, split it into words with `/\s+/`, clear the element, and rebuild content inside one `[data-reveal-line]` wrapper.
- Each word is placed inside `[data-reveal-inner]`.
- `[data-reveal-line]` uses `display: block` and `overflow: hidden`.
- `[data-reveal-inner]` uses `display: inline-block`, starts at `transform: translateY(110%)`, and has `will-change: transform`.
- Add `margin-right: 0.18em` to every `[data-reveal-inner]` except the last to preserve word spacing.
- Animate with:
  - `gsap.to(target.querySelectorAll("[data-reveal-inner]"), { y: 0, duration: 0.9, ease: "power3.out", stagger: 0.025, scrollTrigger: { trigger: target, start: "top 86%", once: true } })`

Stacked card behavior:
- Query sections with `[data-effect="flick-cards-stack"][data-flick-cards-init]`.
- Inside each section:
  - Query `[data-flick-cards-list]`.
  - Query cards with `[data-flick-cards-item]`.
  - Require 8 cards; if fewer than 7, log `Not minimum of 7 cards` and stop drag setup.
  - Start `activeIndex` at `0`.
  - Set `sliderWidth = slider.offsetWidth || window.innerWidth`.
  - Use drag threshold `0.1`.
  - Inject one `[data-flick-cards-dragger]` inside each card.
  - Set `data-flick-drag-status="grab"`.
- Card placement function `getConfig(i, currentIndex)`:
  - Calculate `diff = i - currentIndex`.
  - If `diff > total / 2`, subtract `total`.
  - If `diff < -total / 2`, add `total`.
  - `diff === 0`: `{ x: 0, y: 0, rot: 0, s: 1, o: 1, z: 5 }`
  - `diff === 1`: `{ x: 25, y: 1, rot: 10, s: 0.9, o: 1, z: 4 }`
  - `diff === -1`: `{ x: -25, y: 1, rot: -10, s: 0.9, o: 1, z: 4 }`
  - `diff === 2`: `{ x: 45, y: 5, rot: 15, s: 0.8, o: 1, z: 3 }`
  - `diff === -2`: `{ x: -45, y: 5, rot: -15, s: 0.8, o: 1, z: 3 }`
  - Default: use `dir = diff > 0 ? 1 : -1`, return `{ x: 55 * dir, y: 5, rot: 20 * dir, s: 0.6, o: 0, z: 2 }`.
- Render function:
  - Set card status according to x position: `active`, `2-after`, `2-before`, `3-after`, `3-before`, or `hidden`.
  - Set `card.style.zIndex = cfg.z`.
  - Immediate render uses `gsap.set(card, vars)`.
  - Animated render uses `gsap.to(card, { duration: 0.6, ease: "elastic.out(1.2, 1)", xPercent, yPercent, rotation, scale, opacity })`.
- Draggable setup:
  - `Draggable.create(draggers, { type: "x", edgeResistance: 0.8, bounds: () => ({ minX: -sliderWidth / 2, maxX: sliderWidth / 2 }), inertia: false, ... })`
  - `onPress`: update `sliderWidth`, store pointer X/Y, set drag status to `grabbing`.
  - `onDrag`: compute `rawProgress = this.x / sliderWidth`, `progress = Math.min(1, Math.abs(rawProgress))`, `direction = rawProgress > 0 ? -1 : 1`, and `nextIndex = (activeIndex + direction + total) % total`. Interpolate each card from current config to next config with linear property mixing.
  - `onRelease`: set drag status to `grab`; if `raw > threshold`, shift `-1`; if `raw < -threshold`, shift `1`; update active index and animate stack if shifted. Reset dragger x to `0` with `duration: 0.3`, `ease: "power1.out"`.
  - If release drag distance is under `4`, temporarily disable pointer events and dispatch a click to the element under the pointer.
- Resize behavior:
  - On window resize, update `sliderWidth` and rerender current index immediately.
- Preview helper:
  - On `DOMContentLoaded`, call `initLineReveal()`, `initFlickCardsStack()`, and `ScrollTrigger.refresh()`.
  - If URL has `?preview-block=1`, scroll to the section offset minus `24px`. This is only for preview screenshots and may be omitted in production.
