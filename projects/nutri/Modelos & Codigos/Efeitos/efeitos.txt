## CÓDIGO JAVASCRIPT
Adicione antes do </body> (ou no arquivo JS da página):

```javascript
import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.165.0/build/three.module.js";

class DottedSurface {
  constructor(container, options = {}) {
    this.container = container;
    this.options = {
      separation: 150,
      amountX: 40,
      amountY: 60,
      dotSize: 8,
      waveAmpX: 50,
      waveAmpY: 50,
      speed: 0.1,
      ...options
    };

    this.scene = null;
    this.camera = null;
    this.renderer = null;
    this.geometry = null;
    this.material = null;
    this.points = null;
    this.animationId = null;
    this.count = 0;
    this.isReducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    this.handleResize = this.handleResize.bind(this);
    this.handleMotionPreference = this.handleMotionPreference.bind(this);
    this.handleThemeMutation = this.handleThemeMutation.bind(this);
    this.animate = this.animate.bind(this);
  }

  init() {
    if (!this.container) return;

    const { width, height } = this.getContainerSize();
    if (width === 0 || height === 0) return;

    this.scene = new THREE.Scene();
    this.scene.fog = new THREE.Fog(0xffffff, 2000, 10000);

    this.camera = new THREE.PerspectiveCamera(60, width / height, 1, 10000);
    this.camera.position.set(0, 355, 1220);

    this.renderer = new THREE.WebGLRenderer({
      alpha: true,
      antialias: true
    });
    this.renderer.setPixelRatio(window.devicePixelRatio);
    this.renderer.setSize(width, height);
    this.renderer.setClearColor(this.scene.fog.color, 0);

    this.container.appendChild(this.renderer.domElement);
    this.createPoints();
    this.applyThemeColors();

    window.addEventListener("resize", this.handleResize);

    this.motionQuery = window.matchMedia("(prefers-reduced-motion: reduce)");
    this.motionQuery.addEventListener("change", this.handleMotionPreference);

    this.themeObserver = new MutationObserver(this.handleThemeMutation);
    this.themeObserver.observe(document.documentElement, {
      attributes: true,
      attributeFilter: ["class", "data-theme"]
    });

    if (this.isReducedMotion) {
      this.renderFrame();
      return;
    }

    this.animate();
  }

  getContainerSize() {
    const rect = this.container.getBoundingClientRect();
    return {
      width: Math.max(1, Math.floor(rect.width)),
      height: Math.max(1, Math.floor(rect.height))
    };
  }

  createPoints() {
    const { separation, amountX, amountY, dotSize } = this.options;

    const positions = [];
    const colors = [];

    for (let ix = 0; ix < amountX; ix += 1) {
      for (let iy = 0; iy < amountY; iy += 1) {
        const x = ix * separation - (amountX * separation) / 2;
        const y = 0;
        const z = iy * separation - (amountY * separation) / 2;
        positions.push(x, y, z);
        colors.push(0.08, 0.08, 0.08);
      }
    }

    this.geometry = new THREE.BufferGeometry();
    this.geometry.setAttribute("position", new THREE.Float32BufferAttribute(positions, 3));
    this.geometry.setAttribute("color", new THREE.Float32BufferAttribute(colors, 3));

    this.material = new THREE.PointsMaterial({
      size: dotSize,
      vertexColors: true,
      transparent: true,
      opacity: 0.8,
      sizeAttenuation: true
    });

    this.points = new THREE.Points(this.geometry, this.material);
    this.scene.add(this.points);
  }

  detectDarkTheme() {
    const root = document.documentElement;
    const dataTheme = root.getAttribute("data-theme");
    if (dataTheme) return dataTheme === "dark";

    if (root.classList.contains("dark")) return true;
    return window.matchMedia("(prefers-color-scheme: dark)").matches;
  }

  applyThemeColors() {
    if (!this.geometry) return;

    const isDark = this.detectDarkTheme();
    const color = isDark ? [0.78, 0.78, 0.78] : [0.08, 0.08, 0.08];
    const colorAttr = this.geometry.getAttribute("color");

    for (let i = 0; i < colorAttr.count; i += 1) {
      colorAttr.setXYZ(i, color[0], color[1], color[2]);
    }

    colorAttr.needsUpdate = true;
  }

  updateWave() {
    const { amountX, amountY, waveAmpX, waveAmpY } = this.options;
    const positionAttribute = this.geometry.getAttribute("position");
    const positions = positionAttribute.array;

    let i = 0;
    for (let ix = 0; ix < amountX; ix += 1) {
      for (let iy = 0; iy < amountY; iy += 1) {
        const index = i * 3;
        positions[index + 1] =
          Math.sin((ix + this.count) * 0.3) * waveAmpX +
          Math.sin((iy + this.count) * 0.5) * waveAmpY;
        i += 1;
      }
    }

    positionAttribute.needsUpdate = true;
  }

  renderFrame() {
    if (!this.renderer || !this.scene || !this.camera) return;
    this.renderer.render(this.scene, this.camera);
  }

  animate() {
    this.animationId = requestAnimationFrame(this.animate);
    this.updateWave();
    this.renderFrame();
    this.count += this.options.speed;
  }

  handleResize() {
    if (!this.camera || !this.renderer) return;
    const { width, height } = this.getContainerSize();
    this.camera.aspect = width / height;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(width, height);
    this.renderFrame();
  }

  handleMotionPreference(event) {
    this.isReducedMotion = event.matches;

    if (this.isReducedMotion) {
      if (this.animationId) cancelAnimationFrame(this.animationId);
      this.animationId = null;
      this.renderFrame();
      return;
    }

    if (!this.animationId) this.animate();
  }

  handleThemeMutation() {
    this.applyThemeColors();
    this.renderFrame();
  }

  destroy() {
    window.removeEventListener("resize", this.handleResize);

    if (this.motionQuery) {
      this.motionQuery.removeEventListener("change", this.handleMotionPreference);
    }

    if (this.themeObserver) {
      this.themeObserver.disconnect();
    }

    if (this.animationId) cancelAnimationFrame(this.animationId);

    if (this.points) {
      this.points.geometry.dispose();
      this.points.material.dispose();
      this.scene.remove(this.points);
    }

    if (this.renderer) {
      this.renderer.dispose();
      if (this.renderer.domElement && this.container.contains(this.renderer.domElement)) {
        this.container.removeChild(this.renderer.domElement);
      }
    }
  }
}

function mountDottedSurface() {
  const container = document.querySelector(".dotted-surface");
  if (!container) return null;

  const effect = new DottedSurface(container);
  effect.init();
  return effect;
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", mountDottedSurface, { once: true });
} else {
  mountDottedSurface();
}
```

```javascript
const DEFAULT_OPTIONS = {
  enableClickInteraction: true,
  tubeColors: ["#f967fb", "#53bc28", "#6958d5"],
  lightColors: ["#83f36e", "#fe8a2e", "#ff008a", "#60aed5"],
  lightIntensity: 200,
  randomTubeColorCount: 3,
  randomLightColorCount: 4
};

function randomColors(count) {
  return Array.from({ length: count }, () => {
    const color = Math.floor(Math.random() * 16777215).toString(16).padStart(6, "0");
    return `#${color}`;
  });
}

class NeonFlowEffect {
  constructor(container, options = {}) {
    this.container = container;
    this.canvas = container?.querySelector(".neon-flow-canvas") || null;
    this.options = { ...DEFAULT_OPTIONS, ...options };
    this.app = null;
    this.destroyed = false;
    this.motionQuery = window.matchMedia("(prefers-reduced-motion: reduce)");

    this.handleClick = this.handleClick.bind(this);
    this.handleResize = this.handleResize.bind(this);
  }

  async init() {
    if (!this.container || !this.canvas) return;

    if (this.motionQuery.matches) {
      this.container.dataset.motion = "reduced";
      return;
    }

    try {
      const module = await import(
        "https://cdn.jsdelivr.net/npm/threejs-components@0.0.19/build/cursors/tubes1.min.js"
      );
      const TubesCursor = module.default;
      if (!TubesCursor || this.destroyed) return;

      this.app = TubesCursor(this.canvas, {
        tubes: {
          colors: this.options.tubeColors,
          lights: {
            intensity: this.options.lightIntensity,
            colors: this.options.lightColors
          }
        }
      });

      this.container.dataset.motion = "active";
      this.attachListeners();
    } catch (error) {
      this.container.dataset.motion = "failed";
      console.error("Neon Flow: falha ao carregar efeito", error);
    }
  }

  attachListeners() {
    if (this.options.enableClickInteraction) {
      this.container.addEventListener("click", this.handleClick);
    }
    window.addEventListener("resize", this.handleResize);
  }

  handleResize() {
    if (this.app && typeof this.app.resize === "function") {
      this.app.resize();
    }
  }

  handleClick(event) {
    if (!this.options.enableClickInteraction || !this.app || !this.app.tubes) return;
    if (event.target.closest("[data-neon-flow-ignore-click='true']")) return;

    const tubeColors = randomColors(this.options.randomTubeColorCount);
    const lightColors = randomColors(this.options.randomLightColorCount);

    if (typeof this.app.tubes.setColors === "function") {
      this.app.tubes.setColors(tubeColors);
    }
    if (typeof this.app.tubes.setLightsColors === "function") {
      this.app.tubes.setLightsColors(lightColors);
    }
  }

  destroy() {
    this.destroyed = true;

    this.container?.removeEventListener("click", this.handleClick);
    window.removeEventListener("resize", this.handleResize);

    if (this.app && typeof this.app.destroy === "function") {
      this.app.destroy();
    }

    this.app = null;
  }
}

function mountNeonFlow() {
  const container = document.querySelector(".hero-neon-flow .neon-flow");
  if (!container) return null;

  const effect = new NeonFlowEffect(container);
  effect.init();
  return effect;
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", mountNeonFlow, { once: true });
} else {
  mountNeonFlow();
}
```

```javascript
function mountLampEffect(section) {
  const prefersReducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  if (prefersReducedMotion) {
    section.classList.add("is-visible", "reduced-motion");
    return null;
  }

  if (!("IntersectionObserver" in window)) {
    section.classList.add("is-visible");
    return null;
  }

  const observer = new IntersectionObserver(
    (entries, obs) => {
      entries.forEach((entry) => {
        if (!entry.isIntersecting) return;
        entry.target.classList.add("is-visible");
        obs.unobserve(entry.target);
      });
    },
    {
      threshold: 0.3
    }
  );

  observer.observe(section);
  return observer;
}

function initLampEffects() {
  const sections = document.querySelectorAll(".hero-lamp[data-lamp]");
  if (!sections.length) return;

  const observers = [];
  sections.forEach((section) => {
    const observer = mountLampEffect(section);
    if (observer) observers.push(observer);
  });

  window.addEventListener(
    "beforeunload",
    () => {
      observers.forEach((observer) => observer.disconnect());
    },
    { once: true }
  );
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", initLampEffects, { once: true });
} else {
  initLampEffects();
}
```

## CÓDIGO JAVASCRIPT
Adicione antes do </body> (ou no arquivo JS da página):

```javascript
/**
 * scroll-infinite-panels
 * Paineis fullscreen empilhados com GSAP ScrollTrigger + loop infinito.
 *
 * Dependencias (incluir no <head> ANTES deste script):
 *   <script src="https://cdn.jsdelivr.net/npm/gsap@3/dist/gsap.min.js"></script>
 *   <script src="https://cdn.jsdelivr.net/npm/gsap@3/dist/ScrollTrigger.min.js"></script>
 */
(function () {
  // Aguarda GSAP estar disponivel
  if (typeof gsap === 'undefined' || typeof ScrollTrigger === 'undefined') {
    console.warn('[scroll-infinite-panels] GSAP ou ScrollTrigger nao encontrado. Carregue as dependencias primeiro.');
    return;
  }

  // Respeita preferencia de reducao de movimento
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

  gsap.registerPlugin(ScrollTrigger);

  var wrapper = document.querySelector('.ilp-wrapper');
  if (!wrapper) return;

  var panels = gsap.utils.toArray('.ilp-panel');
  var numPanels = panels.length;
  if (numPanels < 2) return;

  // ---- Aplicar z-index progressivo para empilhamento correto ----
  // O primeiro painel fica embaixo, o ultimo fica em cima
  panels.forEach(function (panel, i) {
    panel.style.zIndex = i + 1;
    // Passa o numero para o pseudo-elemento ::before via dataset
    panel.dataset.ilpNum = String(i + 1).padStart(2, '0');
  });

  // ---- Criar indicadores de progresso (dots laterais) ----
  var progressBar = document.createElement('div');
  progressBar.className = 'ilp-progress-bar';
  progressBar.setAttribute('aria-hidden', 'true');
  panels.forEach(function (_, i) {
    var dot = document.createElement('div');
    dot.className = 'ilp-dot' + (i === 0 ? ' is-active' : '');
    progressBar.appendChild(dot);
  });
  document.body.appendChild(progressBar);

  var dots = progressBar.querySelectorAll('.ilp-dot');

  function setActiveDot(index) {
    dots.forEach(function (dot, i) {
      dot.classList.toggle('is-active', i === index);
    });
  }

  // ---- Pin de cada painel (empilhamento) ----
  // pinSpacing: false garante que cada painel sobrepoe o anterior
  // sem adicionar espaco extra no documento
  panels.forEach(function (panel, i) {
    ScrollTrigger.create({
      trigger: panel,
      start: 'top top',
      pin: true,
      pinSpacing: false,
      onEnter: function () { setActiveDot(i); },
      onEnterBack: function () { setActiveDot(i); },
    });
  });

  // ---- Loop infinito ----
  // Detecta quando chegamos ao final (ultimo painel totalmente visivel)
  // e redireciona para o começo; faz o mesmo no sentido inverso.
  var isJumping = false;

  ScrollTrigger.create({
    trigger: panels[numPanels - 1],
    start: 'top top',
    onEnter: function () {
      // Chegou no ultimo painel rolando para frente → volta ao inicio
      if (isJumping) return;
      isJumping = true;
      // Usa requestAnimationFrame para evitar conflito com o ciclo do ScrollTrigger
      requestAnimationFrame(function () {
        window.scrollTo({ top: 0, behavior: 'instant' });
        ScrollTrigger.refresh();
        setActiveDot(0);
        // Libera após o ciclo de refresh
        setTimeout(function () { isJumping = false; }, 100);
      });
    },
  });

  ScrollTrigger.create({
    trigger: panels[0],
    start: 'top top',
    onLeaveBack: function () {
      // Passou do topo para tras → pula para logo antes do ultimo painel
      if (isJumping) return;
      isJumping = true;
      requestAnimationFrame(function () {
        // Pula para a posicao de scroll do segundo-ultimo painel
        var target = panels[numPanels - 2];
        var targetTop = target.getBoundingClientRect().top + window.scrollY;
        window.scrollTo({ top: targetTop, behavior: 'instant' });
        ScrollTrigger.refresh();
        setActiveDot(numPanels - 2);
        setTimeout(function () { isJumping = false; }, 100);
      });
    },
  });

  // ---- Cleanup ao sair da pagina ----
  window.addEventListener('beforeunload', function () {
    ScrollTrigger.getAll().forEach(function (st) { st.kill(); });
    if (progressBar.parentNode) progressBar.parentNode.removeChild(progressBar);
  });
})();
```

```javascript
/**
 * scroll-horizontal-gallery
 * Galeria horizontal acionada pelo scroll vertical com GSAP ScrollTrigger.
 * Baseado em: https://demos.gsap.com/demo/horizontal-scrolling-gallery/
 *
 * Dependencias obrigatorias (incluir no <head> antes deste script):
 *   <script src="https://cdn.jsdelivr.net/npm/gsap@3/dist/gsap.min.js"></script>
 *   <script src="https://cdn.jsdelivr.net/npm/gsap@3/dist/ScrollTrigger.min.js"></script>
 *
 * ScrollSmoother (opcional, requer GSAP v3 + conta gratuita em https://gsap.com):
 *   <script src="https://cdn.jsdelivr.net/npm/gsap@3/dist/ScrollSmoother.min.js"></script>
 */
(function () {
  if (typeof gsap === 'undefined' || typeof ScrollTrigger === 'undefined') {
    console.warn('[scroll-horizontal-gallery] GSAP ou ScrollTrigger nao encontrado.');
    return;
  }

  // Respeita prefers-reduced-motion (galeria usa scroll nativo no CSS)
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

  gsap.registerPlugin(ScrollTrigger);

  // ---- ScrollSmoother (opcional) ----
  // Descomente o bloco abaixo se tiver o ScrollSmoother disponivel
  // Melhora muito a fluidez do scroll, mas nao e obrigatorio.
  /*
  if (typeof ScrollSmoother !== 'undefined') {
    gsap.registerPlugin(ScrollSmoother);
    ScrollSmoother.create({
      wrapper: '#hsg-smooth-wrapper',
      content: '#hsg-smooth-content',
      smooth: 2,
      normalizeScroll: true,
      ignoreMobileResize: true,
    });
  }
  */

  // ---- Galeria horizontal ----
  var portfolio = document.getElementById('hsg-portfolio');
  if (!portfolio) return;

  var galleryWrappers = gsap.utils.toArray('.hsg-gallery-wrapper');

  galleryWrappers.forEach(function (sec) {
    var strip = sec.querySelector('.hsg-gallery-strip');
    if (!strip) return;

    var pinWrapWidth;
    var horizontalScrollLength;

    function refresh() {
      pinWrapWidth        = strip.scrollWidth;
      horizontalScrollLength = pinWrapWidth - window.innerWidth;
    }

    refresh();

    // Move o strip horizontalmente enquanto o usuario rola verticalmente
    gsap.to(strip, {
      scrollTrigger: {
        scrub: true,                         // sincronizado ao scrollbar
        trigger: sec,
        pin: sec,                            // fixa a secao enquanto anima
        start: 'center center',
        end: function () { return '+=' + pinWrapWidth; },
        invalidateOnRefresh: true,           // recalcula no resize
      },
      x: function () { return -horizontalScrollLength; },
      ease: 'none',
    });

    // Recalcula largura ao redimensionar
    ScrollTrigger.addEventListener('refreshInit', refresh);
  });
})();
```

```javascript
 *
 * Dependencias (CDN):

 *   <script src="https://cdn.jsdelivr.net/npm/gsap@3/dist/SplitText.min.js"></script>

gsap.registerPlugin(SplitText);

/* Guarda todas as instancias de split para o replay */
let activeSplits = [];

/* ---- Funcao principal ---- */
function initMaskedLines() {
  /* Aguarda fontes carregarem para o SplitText calcular corretamente as linhas */
  document.fonts.ready.then(() => {
    activeSplits = [];

    SplitText.create(".tm-split", {
      type: "words,lines",
      linesClass: "line",   /* classe aplicada ao wrapper de cada linha */
      autoSplit: true,       /* re-divide automaticamente ao redimensionar */
      mask: "lines",         /* adiciona overflow:hidden em cada linha (efeito de mascara) */

      onSplit(self) {
        /* Mostra o elemento antes de animar (estava opacity:0 no CSS) */
        gsap.set(self.elements, { opacity: 1 });

        /* Cada linha entra de baixo para cima com stagger */
        const anim = gsap.from(self.lines, {
          yPercent: 100,
          opacity: 0,
          duration: 0.6,
          stagger: 0.1,
          ease: "expo.out",
        });

        activeSplits.push({ self, anim });
        return anim; /* retorno obrigatorio para o autoSplit funcionar com resize */
      },
    });
  });
}

/* ---- Replay em câmera lenta ---- */
const replayBtn = document.querySelector(".tm-replay");
if (replayBtn) {
  replayBtn.addEventListener("click", () => {
    activeSplits.forEach(({ anim }) => {
      anim.timeScale(0.2).play(0);

}
```

```javascript
/**
 * Background — Neon Orbs
 * Injeta as letras animadas no título e gerencia o dark mode manual.
 *
 * Dependencias: nenhuma (CSS puro + este script)
 */

(function () {
  /* ---- Animacao letra-a-letra do titulo ---- */
  document.querySelectorAll(".norb-title[data-norb-text]").forEach(function (el) {
    var text = el.getAttribute("data-norb-text") || "";
    var baseDelay = 800; /* ms — quando os orbs já estão visíveis */
    var perLetter = 50; /* ms entre cada letra */

    text.split("").forEach(function (char, i) {
      var span = document.createElement("span");
      span.textContent = char === " " ? "\u00A0" : char; /* preserva espaços */
      span.style.animationDelay = baseDelay + i * perLetter + "ms";
      el.appendChild(span);
    });
  });

  /* ---- Dark mode manual (toggle programatico, se necessario) ----
     Por padrao o modo e determinado pelo prefers-color-scheme (CSS).
     Para forcar dark/light manualmente, chame:
       NeonOrbs.setDark(true)  — ativa dark
       NeonOrbs.setDark(false) — ativa light
  ---- */
  window.NeonOrbs = {
    setDark: function (dark) {
      document.querySelectorAll(".norb-wrapper").forEach(function (el) {
        el.classList.toggle("dark", dark);
      });
    },
  };
})();
```

Adicione antes do </body> (ou no arquivo JS da página):

```javascript
/**
 * cards-scroll-float
 *
 * Comportamentos:
 *  1. Cards com [data-sfloat-tx] — parallax de scroll:
 *     translateX + rotateZ calculados a partir da posição
 *     do card em relação ao centro da viewport.
 *     - data-sfloat-tx  : velocidade de translação (ex.: "4")
 *     - data-sfloat-rz  : velocidade de rotação    (ex.: "0.3")
 *     - data-sfloat-dir : direção (1 = direita, -1 = esquerda)
 *
 *  2. Cards com [data-sfloat-fadein] — observados por
 *     IntersectionObserver; recebem a classe .sfloat--visible
 *     ao entrar 20% no viewport.
 *
 *  Parallax é desativado em telas ≤ 768 px (mobile).
 */
(function () {
  'use strict';

  /* ── 1. IntersectionObserver: fadeInUp no card central ── */
  var fadeCards = document.querySelectorAll('.sfloat__card[data-sfloat-fadein]');
  if (fadeCards.length && 'IntersectionObserver' in window) {
    var observer = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add('sfloat--visible');
          observer.unobserve(entry.target);
        }
      });
    }, { threshold: 0.2 });

    fadeCards.forEach(function (card) {
      observer.observe(card);
    });
  } else {
    /* fallback: exibe imediatamente */
    fadeCards.forEach(function (card) {
      card.classList.add('sfloat--visible');
    });
  }

  /* ── 2. Scroll parallax nos cards externos ── */
  var floatCards = document.querySelectorAll('.sfloat__card[data-sfloat-tx]');
  if (!floatCards.length) return;

  function isMobile() {
    return window.innerWidth <= 768;
  }

  function updateParallax() {
    if (isMobile()) {
      /* remove transforms no mobile */
      floatCards.forEach(function (card) {
        card.style.transform = '';
      });
      return;
    }

    var vh = window.innerHeight;

    floatCards.forEach(function (card) {
      var rect   = card.getBoundingClientRect();
      var center = rect.top + rect.height / 2;

      /*
       * progress: posição relativa do card ao centro da viewport.
       *   0   → card está exatamente no centro
       *  -0.5 → card está no topo da viewport
       *  +0.5 → card está na base da viewport
       */
      var progress = (vh / 2 - center) / vh;

      var tx  = parseFloat(card.dataset.sfloatTx)  || 0;  /* velocidade X   */
      var rz  = parseFloat(card.dataset.sfloatRz)  || 0;  /* velocidade rot */
      var dir = parseFloat(card.dataset.sfloatDir) || 1;  /* direção        */

      /*
       * Fórmulas calibradas a partir dos parâmetros do Elementor Motion FX:
       *   translateX (px) = dir * tx * progress * 80
       *   rotateZ   (deg) = dir * rz * progress * 10
       *
       * Com tx=4, progress=0.4 → txPx = 128 px  (sutl mas visível)
       * Com rz=0.3, progress=0.4 → rzDeg = 1.2 °  (muito sutil)
       */
      var txPx  = dir * tx  * progress * 80;
      var rzDeg = dir * rz  * progress * 10;

      card.style.transform =
        'translateX(' + txPx.toFixed(2)  + 'px) ' +
        'rotateZ('    + rzDeg.toFixed(3) + 'deg)';
    });
  }

  window.addEventListener('scroll', updateParallax, { passive: true });
  window.addEventListener('resize', updateParallax, { passive: true });
  updateParallax();
})();

```

```javascript
/**
 * background-gradient-bars
 *
 * Para cada elemento [data-gbg] na página, gera dinamicamente
 * as divs das barras dentro de .gbg__bars e aplica a altura e o
 * delay de animação corretos.
 *
 * Configuração via atributos data na <section>:
 *   data-gbg-bars     — número de barras (padrão: 7)
 *   data-gbg-from     — cor base do gradiente (padrão: "rgb(255, 60, 0)")
 *   data-gbg-duration — duração em segundos de cada pulsação (padrão: 2)
 *
 * Perfil de altura:
 *   Barras nas extremidades chegam a 100% de escala;
 *   barras no centro chegam a apenas 30%.
 *   Curva: minHeight + (maxHeight - minHeight) * pow(distFromCenter * 2, 1.2)
 */
(function () {
  'use strict';

  /**
   * Calcula a escala inicial (0–1) de uma barra.
   * @param {number} index - índice da barra (0-based)
   * @param {number} total - total de barras
   * @returns {number} escala entre 0.30 e 1.00
   */
  function calcScale(index, total) {
    var position          = index / (total - 1);
    var maxHeight         = 100;
    var minHeight         = 30;
    var distFromCenter    = Math.abs(position - 0.5);
    var heightPercentage  = Math.pow(distFromCenter * 2, 1.2);
    return (minHeight + (maxHeight - minHeight) * heightPercentage) / 100;
  }

  document.querySelectorAll('.gbg[data-gbg]').forEach(function (section) {
    var numBars  = parseInt(section.dataset.gbgBars || '7', 10);
    var from     = section.dataset.gbgFrom     || 'rgb(255, 60, 0)';
    var duration = parseFloat(section.dataset.gbgDuration || '2');

    var container = section.querySelector('.gbg__bars');
    if (!container) return;

    var inner = document.createElement('div');
    inner.className = 'gbg__bars-inner';

    for (var i = 0; i < numBars; i++) {
      var bar = document.createElement('div');
      bar.className = 'gbg__bar';

      var scale = calcScale(i, numBars);
      bar.style.setProperty('--gbg-scale',    String(scale.toFixed(4)));
      bar.style.setProperty('--gbg-from',     from);
      bar.style.setProperty('--gbg-duration', duration + 's');
      bar.style.animationDelay = (i * 0.1).toFixed(1) + 's';

      inner.appendChild(bar);
    }

    container.appendChild(inner);
  });
})();
```

```javascript
/**
 * Background — Bokeh Particles
 * Gera partículas flutuantes dinamicamente.
 *
 * Uso: Adicionar um container com data-bokeh-count na seção.
 *   <div class="bkh-container" data-bokeh-count="15"></div>
 *
 * Sem dependências externas.
 */

(function () {
  "use strict";

  // Configuração
  var MIN_SIZE     = 2;     // px
  var MAX_SIZE     = 8;     // px
  var MIN_DRIFT    = 10;    // s — duração mínima da animação
  var MAX_DRIFT    = 30;    // s — duração máxima
  var MAX_DELAY    = 5;     // s — delay máximo
  var FLASH_CHANCE = 0.3;   // 30% das partículas terão flash

  function initBokeh() {
    var containers = document.querySelectorAll(".bkh-container[data-bokeh-count]");

    containers.forEach(function (container) {
      var count = parseInt(container.getAttribute("data-bokeh-count") || "15", 10);

      for (var i = 0; i < count; i++) {
        var p = document.createElement("span");
        p.className = "bkh-particle";

        // Tamanho aleatório
        var size = Math.random() * (MAX_SIZE - MIN_SIZE) + MIN_SIZE;
        p.style.width  = size + "px";
        p.style.height = size + "px";

        // Posição aleatória
        p.style.left = Math.random() * 100 + "%";
        p.style.top  = Math.random() * 100 + "%";

        // Duração e delay aleatórios
        p.style.animationDuration = (Math.random() * (MAX_DRIFT - MIN_DRIFT) + MIN_DRIFT) + "s";
        p.style.animationDelay    = (Math.random() * MAX_DELAY) + "s";

        // Chance de flash
        if (Math.random() > (1 - FLASH_CHANCE)) {
          p.classList.add("bkh-particle--flash");
          p.style.animationDuration = (Math.random() * 6 + 4) + "s";
        }

        container.appendChild(p);
      }
    });
  }

  // Boot
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", initBokeh);
  } else {
    initBokeh();
  }
})();
```

## CÓDIGO JAVASCRIPT
Adicione antes do </body> (ou no arquivo JS da página):

```javascript
/**
 * Scroll — Reveal GSAP
 * Anima elementos com [data-anim] ao entrarem na viewport.
 * Usa IntersectionObserver + GSAP fromTo().
 *
 * Dependência: GSAP 3.x deve estar carregado antes deste script.
 *   <script src="https://cdn.jsdelivr.net/npm/gsap@3/dist/gsap.min.js"></script>
 *
 * Tipos disponíveis:
 *   data-anim="fade-up"     → opacity 0→1, y 30→0
 *   data-anim="fade-left"   → opacity 0→1, x -40→0
 *   data-anim="fade-right"  → opacity 0→1, x 40→0
 *   data-anim="fade-in"     → opacity 0→1
 *   data-anim="zoom-in"     → opacity 0→1, scale 0.9→1
 *   data-anim="slide-up"    → opacity 0→1, y 50→0 (ease power3)
 *
 * Atributos opcionais:
 *   data-anim-delay="0.2"      → delay em segundos
 *   data-anim-duration="0.8"   → duração em segundos
 */

(function () {
  "use strict";

  if (typeof gsap === "undefined") {
    console.warn("[scroll-reveal-gsap] GSAP não encontrado. Elementos ficarão invisíveis.");
    // Fallback: mostrar tudo
    document.querySelectorAll("[data-anim]").forEach(function (el) {
      el.style.opacity = "1";
    });
    return;
  }

  function initScrollReveals() {
    var reveals = document.querySelectorAll("[data-anim]");
    if (!reveals.length) return;

    var observer = new IntersectionObserver(
      function (entries) {
        entries.forEach(function (entry) {
          if (!entry.isIntersecting) return;

          var el = entry.target;
          var animType = el.getAttribute("data-anim");
          var delay    = parseFloat(el.getAttribute("data-anim-delay") || "0");
          var duration = parseFloat(el.getAttribute("data-anim-duration") || "0.8");

          var from = {};
          var to   = {
            opacity: 1,
            duration: duration,
            delay: delay,
            ease: "power2.out"
          };

          switch (animType) {
            case "fade-up":
              from = { opacity: 0, y: 30 };
              break;
            case "fade-left":
              from = { opacity: 0, x: -40 };
              break;
            case "fade-right":
              from = { opacity: 0, x: 40 };
              break;
            case "fade-in":
              from = { opacity: 0 };
              break;
            case "zoom-in":
              from = { opacity: 0, scale: 0.9 };
              break;
            case "slide-up":
              from = { opacity: 0, y: 50 };
              to.ease = "power3.out";
              break;
            default:
              from = { opacity: 0, y: 20 };
          }

          gsap.fromTo(el, from, to);
          el.classList.add("sr-visible");
          observer.unobserve(el);
        });
      },
      {
        threshold: 0.15,
        rootMargin: "0px 0px -50px 0px"
      }
    );

    reveals.forEach(function (el) {
      observer.observe(el);
    });
  }

  // Boot
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", initScrollReveals);
  } else {
    initScrollReveals();
  }
})();

```

## CÓDIGO JAVASCRIPT
Adicione antes do </body> (ou no arquivo JS da página):

```javascript
/**
 * Scroll — Story Text Swap
 * Controla a troca de frases com base na posicao do scroll.
 *
 * Funciona assim:
 * 1. Calcula o progresso do scroll dentro da secao (0 → 1)
 * 2. progress < 0.45 → frase 1 ativa
 * 3. progress >= 0.45 → frase 1 sai, frase 2 entra
 *
 * Para adicionar mais frases, criar novos thresholds e estados.
 */
(function () {
  'use strict';

  function initStoryTextSwap() {
    var section = document.querySelector('.sts-section');
    if (!section) return;

    var phrase1 = document.getElementById('sts-phrase-1');
    var phrase2 = document.getElementById('sts-phrase-2');
    if (!phrase1 || !phrase2) return;

    var currentState = 0; // 0 = nenhuma, 1 = frase 1, 2 = frase 2

    function setState(state) {
      if (state === currentState) return;
      currentState = state;

      phrase1.classList.toggle('sts-phrase--active', state === 1);
      phrase1.classList.toggle('sts-phrase--exit', state === 2);
      if (state !== 2) phrase1.classList.remove('sts-phrase--exit');

      phrase2.classList.toggle('sts-phrase--active', state === 2);
      phrase2.classList.remove('sts-phrase--exit');

      if (state === 0) {
        phrase1.classList.remove('sts-phrase--active', 'sts-phrase--exit');
        phrase2.classList.remove('sts-phrase--active');
      }
    }

    function onScroll() {
      var rect = section.getBoundingClientRect();
      var scrolled = -rect.top;
      var range = section.offsetHeight - window.innerHeight;

      if (scrolled < 0 || range <= 0) {
        setState(0);
        return;
      }

      var progress = scrolled / range; // 0 → 1
      if (progress < 0.45) {
        setState(1);
      } else {
        setState(2);
      }
    }

    window.addEventListener('scroll', onScroll, { passive: true });
    onScroll();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initStoryTextSwap);
  } else {
    initStoryTextSwap();
  }
})();

```

```javascript
/**
 * Scroll — Steps BG Morph
 * Muda a classe de fase da secao conforme os paineis entram na viewport.
 *
 * Funciona assim:
 * 1. IntersectionObserver (threshold 0.5) monitora cada .sbm-panel
 * 2. Quando um painel entra, le seu data-phase
 * 3. Remove todas as classes de fase e adiciona a nova
 * 4. CSS transiciona o fundo e cores automaticamente
 */
(function () {
  'use strict';

  function initStepsBgMorph() {
    var section = document.querySelector('.sbm-section');
    if (!section) return;

    var panels = section.querySelectorAll('.sbm-panel');
    if (!panels.length) return;

    var panelObserver = new IntersectionObserver(
      function (entries) {
        entries.forEach(function (entry) {
          if (entry.isIntersecting) {
            var phase = entry.target.dataset.phase || '1';
            section.classList.remove('sbm--phase-1', 'sbm--phase-2', 'sbm--phase-3');
            section.classList.add('sbm--phase-' + phase);
          }
        });
      },
      { threshold: 0.5 }
    );

    panels.forEach(function (panel) {
      panelObserver.observe(panel);
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initStepsBgMorph);
  } else {
    initStepsBgMorph();
  }
})();
```

```javascript
/**
 * Scroll — Reveal Vanilla
 * IntersectionObserver que adiciona classe .srv-visible
 * quando elementos com [data-anim] entram na viewport.
 * Zero dependencias — JavaScript vanilla puro.
 */
(function () {
  'use strict';

  function initScrollReveal() {
    var elements = document.querySelectorAll('[data-anim]');
    if (!elements.length) return;

    var observer = new IntersectionObserver(
      function (entries) {
        entries.forEach(function (entry) {
          if (entry.isIntersecting) {
            entry.target.classList.add('srv-visible');
            observer.unobserve(entry.target);
          }
        });
      },
      {
        threshold: 0.1,
        rootMargin: '0px 0px -60px 0px'
      }
    );

    elements.forEach(function (el) {
      observer.observe(el);
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initScrollReveal);
  } else {
    initScrollReveal();
  }
})();
```
